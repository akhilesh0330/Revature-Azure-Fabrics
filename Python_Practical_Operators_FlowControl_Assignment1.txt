Q12. Simulate payment retry logic where the system retries payment a maximum of 3 times if payment fails.

# Simulate payment retry logic

max_retries = 3
attempt = 0
payment_success = False

while attempt < max_retries:
    attempt += 1
    print(f"Payment attempt {attempt}")

    # Simulate payment failure
    payment_success = False

    if payment_success:
        print("Payment successful")
        break
    else:
        print("Payment failed")

if not payment_success:
    print("Maximum retry limit reached. Payment unsuccessful.")


Q13. Write a program that keeps checking order status until status becomes 'Delivered' or maximum 5 attempts are reached.

# Order status checking simulation

status = "Processing"
attempts = 0
max_attempts = 5

while attempts < max_attempts:
    attempts += 1
    print("Attempt", attempts, "- Status:", status)

    # Suppose status changes after 3 attempts
    if attempts == 3:
        status = "Delivered"

    if status == "Delivered":
        print("Order delivered successfully!")
        break

if status != "Delivered":
    print("Order not delivered after 5 attempts.")

PART E: Mini Real-Time Challenge
Q15. You are given a list of orders with order_id and amount.
- Skip cancelled orders
- Classify remaining orders as Premium or Regular
- Count total premium orders

# List of orders: (order_id, amount, status)
orders = [
    (101, 50000, "completed"),
    (102, 120000, "completed"),
    (103, 70000, "cancelled"),
    (104, 250000, "completed"),
    (105, 30000, "completed"),
    (106, 150000, "cancelled")
]

premium_count = 0

for order_id, amount, status in orders:

    # Skip cancelled orders
    if status == "cancelled":
        continue

    # Classify order
    if amount > 100000:
        print(order_id, "- Premium Order")
        premium_count += 1
    else:
        print(order_id, "- Regular Order")

print("Total Premium Orders:", premium_count)
